# Ceilings stop a surprise major release from breaking a build.
#
# The aiohttp and zeroconf floors are *security* floors, not compatibility
# ones: those two parse input that comes from off-process — guests over HTTP
# and mDNS from the local network — so an existing checkout must not sit on an
# old copy forever. start.bat re-checks these on every launch, and CI installs
# fresh for every release. Raise them whenever upstream ships a security fix.
PySide6>=6.7,<7
python-osc>=1.9,<2
zeroconf>=0.150,<1
aiohttp>=3.14.3,<4
openvr>=1.23,<3

# one-file exe can't find it. The browser's half (tools/web_guest) is built
# against the iroh these bindings carry (1.2.0 = iroh 1.4.1): bump together.
iroh>=1.2.0,<1.3
